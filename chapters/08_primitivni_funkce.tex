\section{Primitivní funkce}
\subsection{Základní vlastnosti}
\begin{definice}
Nechť $I$ je otevřený interval a $f : I \to \mathbb{R}$. Řekneme, že funkce $F : I \to \mathbb{R}$
je \textbf{primitivní funkcí} k funkci $f$ na $I$, jestliže pro každé $x \in I$ existuje $F'(x)$ a platí $F'(x) = f(x)$.
\end{definice}

\begin{poznamka}
	\,
	\begin{enumerate}[(1)]
		\item Existují funkce, které nemají funkci primitivní, např. $\sgn$ na $\mathbb R $.
		\item Je-li funkce $F$ primitivní k funkci $f$ na $I$, potom $F$ je spojitá na $I$.
		\item Hledání primitivní funkce označujeme jako \textbf{integraci}. Primitivní funkci někdy říkáme
		\textbf{neurčitý integrál}.
		\item Je-li $F$ primitivní k $f$ na $I$, potom $F+c$, kde $c \in \mathbb R $, je také
		primitivní k $f$ na $I$.
	\end{enumerate}
\end{poznamka}

\begin{veta}\label{veta8.1}
Nechť $I$ je otevřený interval, $f, F, G : I \to \mathbb{R}$ jsou funkce, přičemž $F$ a $G$ jsou primitivní funkce k $f$ na $I$. Pak existuje $c \in \mathbb{R}$ takové, že
$F = G + c$.
\end{veta}

\begin{proof}
	Položme
	$$
	    H(x)=F(x)-G(x), \qquad x \in I.
	$$
	Potom pro každé $x\in I$ platí
	$$
	    H^\prime(x) = F^\prime(x) - G^\prime(x) = f(x)-f(x)=0.
	$$
	Odtud máme, že $H(x)$ je konstantní na $I$.
\end{proof}

\begin{oznaceni}
Množinu všech primitivních funkcí k funkci $f$ označujeme symbolem
\[
\int f(x) \d x.
\]
Pro popis této množiny budeme používat značení
\[
\int f(x) \d x \stackrel{c}{=} F(x), \qquad x \in I.
\]
Jednotlivé části symbolu $\int f(x) \d x$ jsou znak integrálu $\int$, integrand $f(x)$ a symbol $\d x$ označující proměnnou, vzhledem k níž integrujeme.
\end{oznaceni}

\paragraph{Tabulkové integrály.}
\begin{align*}
&\int x^n \d x \stackrel{c}{=} \frac{x^{n+1}}{n+1}, \quad x \in \mathbb R  \quad \text{pro } n \in \mathbb{Z},\ n \ge 0;x \in (-\infty,0)\text{ nebo } (0,\infty)\text{ pro }n \in \mathbb Z , n < -1\\
&\int x^\alpha \d x \stackrel{c}{=} \frac{x^{\alpha+1}}{\alpha+1}, \quad x \in (0,\infty) \quad \text{pro } \alpha \in \mathbb{R} \smallsetminus \mathbb{Z}\\
&\int \frac{1}{x}\d x \stackrel{c}{=} \log |x|, \quad x \in (-\infty,0)\text{ nebo }x \in (0,\infty) \\
&\int e^x \d x \stackrel{c}{=} e^x, \quad x \in \mathbb R \\
&\int \sin x \d x \stackrel{c}{=} -\cos x,\quad x \in \mathbb R \\
&\int \cos x \d x \stackrel{c}{=} \sin x,\quad x \in \mathbb R\\
&\int \frac{1}{\cos^2 x}\d x \stackrel{c}{=} \tg x, \quad x \in (-\pi/2+k\pi, \pi/2+k\pi),\ k \in \mathbb Z \\
&\int \frac{-1}{\sin^2 x}\d x \stackrel{c}{=} \cotg x, \quad x \in (k\pi, \pi+k\pi),\ k \in \mathbb Z \\
&\int \frac{1}{\sqrt{1-x^2}}\d x \stackrel{c}{=} \arcsin x, \quad x \in (-1,1)\\
&\int -\frac{1}{\sqrt{1-x^2}}\d x \stackrel{c}{=} \arccos x,\quad x \in (-1,1)\\
&\int \frac{1}{1+x^2}\d x \stackrel{c}{=} \arctg x,\quad x \in \mathbb R\\
&\int -\frac{1}{1+x^2}\d x \stackrel{c}{=} \arccotg x,\quad x \in \mathbb R
\end{align*}

\begin{veta}\label{veta8.2}
Nechť $f$ je spojitá funkce na otevřeném neprázdném intervalu $I$. Pak $f$ má na $I$ primitivní funkci.
\end{veta}

\begin{veta}\label{veta8.3}
Nechť $\alpha \in \mathbb{R} \smallsetminus \{0\}$ a funkce $f$ a $g$ mají na otevřeném intervalu $I$ primitivní funkce. Pak mají funkce $\alpha f$ a $f+g$ primitivní funkce na $I$ a platí
\[
\int (f+g) = \int f + \int g,\qquad
\int \alpha f = \alpha \int f.
\]
\end{veta}

\begin{proof}Obě rovnosti dokážeme zvlášť.
\begin{itemize}
	\item První rovnost
	\begin{enumerate}
		\item[$\subseteq$] Označme $H\in \int(f+g)$, Zvolme $F \in \in f, G\in \int g$. Pak je
		\begin{align*}
		H &= (H-F-G)+F+G, \qquad \text{a}\\
		(H-F-G)^\prime + F^\prime &= f+g-f-g+f = f.
		\end{align*}
		\item[$\supseteq$] Označme $H\in \int f + \int g$. Je
		\begin{align*}
		H &= F_1+G_1, & F_1 \in \int f, \qquad G_1\in \int g\\
		H^\prime &= F^\prime _1+ G_1^\prime = f+g,
		\end{align*}
		a tedy $(H-F-G) +F \in \int f$ a $G\in \int g$.
	\end{enumerate}
	\item Druhá rovnost
	\begin{enumerate}
		\item[$\subseteq$] Vezměme $H\in \int \alpha f$. Pak je $H=\alpha \frac{1}{\alpha}H$, a tedy
		$$
  \left( \frac{1}{\alpha}H \right )^\prime = \frac{1}{\alpha}\alpha f = f,
		$$
		a proto $\frac{1}{\alpha}H \in \int f$.
		\item[$\supseteq$] Vezměme $F \in \int f$. Pak je
		$$
		    (\alpha F)^\prime = \alpha f \in \int \alpha f.
		$$
		\qedhere
	\end{enumerate}
\end{itemize}
\end{proof}

\begin{veta}[Integrace per partes]\label{veta8.4}
Nechť funkce $f$ a $g$ mají na otevřeném intervalu $I$ po řadě primitivní funkce $F$ a $G$ a $f$ je spojitá na $I$. Pak platí
\[
\int fG = FG - \int Fg \quad \text{na } I.
\]
\end{veta}

\begin{proof}
	Dokážeme dvě inkluze.
	\begin{enumerate}
		\item[$\subseteq$] Funkce $f$ i $G$ jsou spojité na $I$. Proto je i $fG$ spojitá
		a podle věty \ref{veta8.3} má primitivní funkci, označme ji $L$, je tedy $L\in \int fG$.
		Je
		$$
		    L = FG-FG+L = FG- ( FG-L).
		$$
		Chceme ověřit, že $(FG-L) \in \int Fg$. Počíáme
		$$
		    (FG-L)^\prime = F^\prime G + FG^\prime - L^\prime = fG+Fg - fG = Fg,
		$$
		což jsme chtěli dokázat.
		\item[$\supseteq$] Již víme, že $Fg$ má primitivní funkci na $I$.  Vezměme si $U\in \int Fg$.
		Chceme dokázat, že
		$
		    FG-U \in \int fG
		$.
		Je
		$$
		    (FG-U)^\prime = F^\prime G + FG^\prime - U^\prime = fG+Fg-Fg=fG,
		$$
		což jsme chtěli dokázat. \qedhere
	\end{enumerate}
\end{proof}

\begin{priklad}\,
\begin{enumerate}[(1)]
	\item Najděte $\int\log x \d x$.
	\par Funkci přepíšeme jako $\int 1\cdot \log x \d x.$ Pak je
	$$
	    \int 1\cdot \log x \d x = x\log x - \int x \cdot \frac{1}{x} \d x = x\log x - \int 1 \d x \stackrel{c}{=} x\log x -x, \quad x \in (0,\infty).
	$$
	\item Najděte $\int \frac{1}{(1+x^2)^n}\d x$ pro nějaké $n \in \mathbb N $.
	\par Označme $I_n = \int \frac{1}{(1+x^2)^n}\d x$. Pak je
	\begin{align*}
	I_n &= \int \underbrace{1}_{f}\cdot \underbrace{\frac{1}{(1+x^2)^n}}_{G}\d x = \underbrace{x}_{F}\cdot \underbrace{\frac{1}{(1+x^2)^n}}_{G} - \int \underbrace{x}_{F} \underbrace{(-n)\frac{1}{(1+x^2)^{n+1}}2x}_{g}\d x\\
&= \frac{x}{(1+x^2)^n}+ 2n \int \frac{x^2}{(1+x^2)^{n+1}}\d x\\
&= \frac{x}{(1+x^2)^n} + 2n \left( \int \frac{1+x^2}{(1+x^2)^{n+1}}\d x - \int \frac{1}{(1+x^2)^{n+1}}\d x \right )\\
&= \frac{x}{(1+x^2)^n} + 2n \left( \int \frac{1}{(1+x^2)^n}\d x - \int \frac{1}{(1+x^2)^{n+1}}\d x \right )\\
&= \frac{x}{(1+x^2)^n} + 2nI_n - 2nI_{n+1}
\end{align*}
a po vyjádření máme
\begin{align*}
	I_{n+1} &= \frac{x}{(1+x^2)^n} + \frac{2n-1}{2n} I_n\\
	I_1 &= \int \frac{1}{1+x^2}\d x \stackrel{c}{=} \arctg x, \quad x \in \mathbb R
	I_2 &\stackrel{c}{=} \frac{x}{2(1+x^2)}+\frac{1}{2}\arctg x, \quad x \in \mathbb R\\
	& \dots
\end{align*}
\item Najděte $\int x \sin x \d x$.
\par Je
\begin{align*}
\int \underbrace{e^x}_{f}\underbrace{\sin x}_{G} \d x &= \underbrace{e^x}_{F}\underbrace{\sin x }_{G} - \int \underbrace{e^x}_{F}\underbrace{\cos x}_g \d x\\
&= e^x - \sin x - \left( e^x\cos x - \int e^x (-\sin x)\d x\right )\\
&= e^x(\sin x -\cos x) - \int e^x \sin x \d x,
\end{align*}
a tedy
$$
    \int e^x \sin x \d x \stackrel{c}{=} \frac{1}{2} e^x (\sin x - \cos x), \quad x \in \mathbb R.
$$
\end{enumerate}
\end{priklad}

\begin{veta}[Darboux] \label{veta8.5}
Nechť $f$ má na otevřeném intervalu $I$ primitivní funkci. Potom $f$ zobrazuje každý interval $J \subset I$ opět na interval.
\end{veta}

\begin{proof}
    Nechť $J\subset I$ je interval. Chceme ukázat, že $f(J)$ je interval.
    Vezměme $y_1, y_2 \in f(J)$, $y_1 < y_2$ a $z\in (y_1,y_2)$. Nalezneme
    $x_1, x_2 \in J$ takové, že $f(x_1) = y_1$, $f(x_2)=y_2$. Předpokládejme, že
    $x_1 < x_2$. Druhý případ by se řešil analogicky. Označme $F$ funkci primitivní k $f$ na $I$.
    Položme
    $$
        H(x) = F(x)-zx, \qquad x \in I.
    $$
    Funkce $H$ je spojitá na $I$, a tedy i na $[x_1,x_2]$. Existuje $x^*\in [x_1,x_2]$, kde
    $H$ nabývá minima na $[x_1,x_2]$. Platí
    $$
    H^\prime(x) = F^\prime(x)-z=f(x)-z, \qquad x \in I,
    $$
    tedy
    \begin{align}
    H^\prime(x_1) &= f(x_1)-z = y_1 - z < 0,\label{rovniceder1:1}\\
    H^\prime(x_2) &= f(x_2)-z = y_2 - z > 0.\label{rovniceder1:2}
    \end{align}
    Z (\ref{rovniceder1:1}) plyne, že existuje $\delta_1 > 0$ takové, že
    $$
        \forall x \in (x_1, x_1+\delta_1): H(x_1)>H(x).
    $$
    Z (\ref{rovniceder1:2}) plyne, že existuje $\delta_2>0$ takové, že
    $$
        \forall x \in (x_2-\delta_2, x_2):H(x)<H(x_2).
    $$
    Odtud plyne, že $x^* \in (x_1,x_2)$. Proto je $x^*$ bodem lokálního minima $H$,
    a tedy $H^\prime(x^*)=0$. Proto je $f(x^*)=z$, což jsme chtěli dokázat.
\end{proof}

\begin{veta}[První věta o substituci] \label{veta8.6}
Nechť $a,b,\alpha,\beta \in \mathbb{R}^*$, $a<b$, $\alpha<\beta$, $F$ je primitivní funkce k $f$ na $(a,b)$,
$\varphi : (\alpha,\beta) \to (a,b)$ a pro každé $t \in \left(\alpha, \beta \right ) $ existuje vlastní
$\varphi^\prime(t)$. Potom platí
\[
\int f(\varphi(t))\, \varphi'(t)\d t \stackrel{c}{=} F(\varphi(t)), \qquad t \in (\alpha,\beta).
\]
\end{veta}

\begin{proof}
	Platí
	$$
	    (F(\varphi(t)))^\prime = F^\prime(\varphi(t))\cdot \varphi^\prime(t) = f(\varphi(t))\cdot \varphi^\prime(t), \qquad t \in (\alpha, \beta).
	$$
\end{proof}

\begin{veta}[Druhá věta o substituci] \label{veta8.7}
Nechť $a,b,\alpha,\beta \in \mathbb{R}^*$, $a<b$, $\alpha<\beta$, $\varphi : (\alpha,\beta) \to (a,b)$,
pro každé $t \in(\alpha, \beta)$ existuje $\varphi^\prime(t)$ vlastní a nenulová a
$\varphi((\alpha, \beta)) = (a,b)$. Nechť $f$ je funkce definovaná na intervalu $(a,b)$ a platí
\[
\int f(\varphi(t)) \cdot\varphi'(t)\d t \stackrel{c}{=} G(t), \qquad t \in (a,b).
\]
Pak platí
\[
\int f(x)\d x \stackrel{c}{=} G(\varphi^{-1}(x)), \qquad x \in (a,b).
\]
\end{veta}

\begin{proof}
	Podle věty \ref{veta8.5} je $\varphi^\prime$ na $(\alpha, \beta)$ kladná nebo je
	$\varphi^\prime$ na $(\alpha, \beta)$
	záporná. Předpokládejme první možnost, druhá se dokáže analogicky. Potom je $\varphi$ rostoucí na
	$(\alpha, \beta)$, a je tedy prostá. Potom $\varphi^{-1}: (a,b)\to (\alpha, \beta)$ je dobře definována.
	Počítejme
	\begin{align*}
	\left( G\circ \varphi^{-1} \right )^\prime &= G^\prime \left(\varphi^{-1}(x)\right)\cdot \left(\varphi^{-1}(x)\right)^\prime\\
&= f \left( \varphi(\varphi^{-1}(x)) \right )\cdot \varphi^\prime \left( \varphi^{-1}(x) \right )\cdot \frac{1}{\underbrace{\varphi^\prime \left( \varphi^{-1}(x) \right ) }_{\ne 0}}.
\end{align*}
\end{proof}

	$$

\end{priklad}

\subsection*{Integrace racionálních funkcí}
\begin{definice}
\textbf{Racionální funkcí} rozumíme reálnou funkci, která je podílem dvou polynomů, přičemž jmenovatel není identicky roven nule.
\end{definice}

\paragraph{Integrace parciálních zlomků.} Rozlišujeme parciální zlomky více typů.
\begin{enumerate}[(a)]
	\item Parciální zlomky prvního typu jsou zlomky tvaru
	$$\int \frac{A}{(x-a)^n}\d x, \quad a \in \mathbb R , A\in \mathbb R , n \in \mathbb N .$$
	Integrujeme je následovně:
	$$
	    \int \frac{A}{(x-a)^n}\d x\stackrel{c}{=}\begin{cases}
						A\log |x-a|, \, x \in (-\infty, a) \text{ nebo } x \in (a,\infty) & n=1,\\
						\frac{1}{1-n}\cdot \frac{A}{(x-a)^{n-1}}, \, x \in (-\infty, a) \text{ nebo } x \in (a,\infty) & n > 1.
					\end{cases}
	$$
	\item Parciální zlomky druhého typu jsou zlomky tvaru
	$$
	    I=\int \frac{Bx+C}{(x^2+\alpha x + \beta)^q}\d x,
	$$
	kde $B,C,\alpha, \beta \in \mathbb R , q \in \mathbb N $ a $\alpha^2-4\beta < 0$ (tedy
	polynom ve jmenovateli nemá reálné kořeny). Pak je:
	\begin{align*}
	I &= \frac{B}{2}\underbrace{\int \frac{2x+\alpha}{(x^2+\alpha x + \beta)^q}\d x}_{I_1} + \left( C-\frac{B\alpha}{2} \right ) \underbrace{\int \frac{1}{(x^2+\alpha x+\beta)^q}\d x}_{I_2}.
\end{align*}
Jednotlivé funkce zintegrujeme následovně:
\begin{align*}
I_1 &\stackrel{c}{=} \begin{cases}
\log \left( x^2+\alpha x + \beta \right ),\,  x \in \mathbb R , & q=1,\\
\frac{1}{1-q}\frac{1}{(x^2+\alpha x +\beta)^{q-1}}, \, x \in \mathbb R, & q > 1.
\end{cases}\\
I_2 &= \int \frac{1}{\left(\left( x+ \frac{\alpha}{2} \right )^2+\beta - \frac{\alpha^2}{4} \right)^q}\d x\\
&= \frac{1}{\left( \beta - \frac{\alpha^2}{4} \right )^q }\int \frac{1}{\left( \left( \frac{x+\frac{\alpha}{2}}{\sqrt{\beta - \frac{\alpha^2}{4}}} \right )^2+1  \right )^q }\d x,
\end{align*}
a tento případ převedeme substitucí na integraci
$$
    \int \frac{1}{(t^2+1)^q}\d t
$$
až na nějakou konstantu. Zvolme
$$
    \varphi(x) = \frac{x+ \frac{\alpha}{2}}{\sqrt{\beta - \frac{\alpha^2}{4}}}.
$$
Pak je
$$
    \varphi^\prime(x) = \frac{1}{\sqrt{\beta - \frac{\alpha^2}{4}}}
$$
a můžeme přepsat
$$
    \int \frac{1}{\left( \left( \frac{x+\frac{\alpha}{2}}{\sqrt{\beta - \frac{\alpha^2}{4}}} \right )^2+1  \right )^q }\d x = \sqrt{\beta - \frac{\alpha^2}{4}}\int \frac{1}{\left( \left( \frac{x+\frac{\alpha}{2}}{\sqrt{\beta - \frac{\alpha^2}{4}}} \right )^2+1  \right )^q } \frac{1}{\sqrt{\beta - \frac{\alpha^2}{4}}}\d x,
$$
a tedy počítáme integrál
$$
f(z)= \frac{1}{\left( z^2+1 \right )^q },
$$
což už umíme.
\end{enumerate}

\begin{veta}\label{veta8.8}
Nechť
$
P(x) = a_n x^n + \cdots + a_1 x + a_0
$
je polynom stupně $n$ s reálnými koeficienty. Pak existují reálná čísla $x_1,\dots,x_k$, $\alpha_1,\dots,\alpha_l$, $\beta_1,\dots,\beta_l$ a přirozená čísla $p_1,\dots,p_k$, $q_1,\dots,q_l$ taková, že
\begin{itemize}
	\item platí
	\[
P(x) = a_n (x-x_1)^{p_1}\cdots(x-x_k)^{p_k}
(x^2+\alpha_1 x+\beta_1)^{q_1}\cdots(x^2+\alpha_l x+\beta_l)^{q_l},
\]
\item žádné dva z polynomů $x-x_1, \dots, x-x_k, x^2+\alpha_1x+\beta_1, \dots, x^2+\alpha_lx+\beta_l$
nemají společný kořen,
\item polynomy $x^2+\alpha_1x+\beta_1, \dots, x^2+\alpha_lx+\beta_l$ nemají žádný reálný kořen.
\end{itemize}
\end{veta}

\begin{veta}[Rozklad na parciální zlomky]\label{veta8.9}
Nechť $P,Q$ jsou polynomy s reálnými koeficienty, takové, že $\operatorname{st} P < \operatorname{st} Q$ a nechť
$$
Q(x) = a_n(x-x_1)^{p_1}\dots (x-x_k^{p_k})\left( x^2+\alpha_1x+\beta_1 \right )^{q_1}\dots \left( x^2+\alpha_l x + \beta_l \right )^{q_l}
$$
je rozklad polynomu $Q$ z věty \ref{veta8.8}. Pak existují jednoznačně určená čísla $A_1^1,\dots,A_{p_1}^1,\dots, A_1^{k},\dots,A_{p_k}^k$,
$B_1^1,C_1^1\dots, B_{q_1}^1C_{q_1}^1,\dots,B_1^l,C_1^l,B_{q_l}^lC_{q_l}^l$ taková, že platí
\begin{align*}
\frac{P(x)}{Q(x)} &= \frac{A_1^1}{(x-x_1)}+\dots + \frac{A_{p_1}^1}{(x-x_1)^{p_1}} + \dots + \frac{A_1^k}{(x-x_k)}+ \dots + \frac{A_{p_k}^k}{(x-x_k)^{p_k}}\\
&+ \frac{B_1^1 x+C_1^1}{(x^2+\alpha_1 x+\beta_1)}+\dots + \frac{B_{q_1}^1x+C_{q_1}^1}{(x^1+\alpha_1 x+\beta_1)^{q_1}}\\
&+  \dots \\
&+ \frac{B_1^l x+C_1^l}{(x^2+\alpha_l x+\beta_l)} + \dots + \frac{B_{q_l}^l x+C_{q_l}^l}{(x^2+\alpha_l x+\beta_l)^{q_l}}
\end{align*}
\end{veta}

\paragraph{Postup při integraci racionální lomené funkce.} Úloha: Spočíteje $\int R(x) \d x$, kde
$$
    R(x) = \frac{P(x)}{Q(x)} = P_1(x) + \frac{P_2(x)}{Q(x)},
$$
přičemž $P(x), P_1(x), P_2(x), Q(x)$ jsou polynomy, $\operatorname{st} P_2(x) < \operatorname{st} Q(x)$.
Dále rozložíme $\frac{P_2(x)}{Q(x)}$ na parciální zlomky a konečně zintegrujeme polynom
$P_1(x)$ a jednotlivé parciální zlomky.

\begin{priklad}
	Spočtěte $\int\frac{x^2+1}{(x+1)(x^2+x+1)}\d x$.
	\par Podle věty \ref{veta8.9} existují $A,B,C \in \mathbb R $ taková, že
	$$
	    \frac{x^2+1}{(x+1)(x^2+x+1)} = \frac{A}{x+1}+ \frac{Bx+C}{x^2+x+1}, \qquad x \in \mathbb R \smallsetminus \left \{ -1 \right \}.
	$$
	Tuto rovnost přepíšeme na
	$$
	    x^2+1 = A(x^2+x+1) + (Bx+C)(x+1), \qquad x \in \mathbb R \smallsetminus \left \{ -1 \right \}.
	$$
	Protože se tyto dvě funkce shodují na prstencovém okolí bodu $-1$ a jsou spojité,
	platí druhá rovnost dokonce na $\mathbb R $. Upravíme
	$$
	    x^2+1 = (A+B)x^2+(A+B+C)x+A+C, \qquad x \in \mathbb R.
	$$
	Metodou porovnání koeficientů dostaneme
	\begin{align*}
		&x^2: 1 = A+B,\\
		&x^1: 0=A+B+C, \\
		&x^0: 1 = A+C.
	\end{align*}
	Lze postupovat jednodušeji, třeba pro $x=1$ dostaneme $A=2$. Pak $B=-1$ a $C = -1$. Máme tedy
	$$
	    \int \frac{x^2+1}{(x+1)(x^2+x+1)}\d x = \underbrace{\int \frac{2}{x+1} \d x}_{I_1}- \underbrace{\int \frac{x+1}{x^2+x+1}\d x}_{I_2}.
	$$
	Spočteme
	\begin{align*}
	I_1 &\stackrel{c}{=} 2\log |x+1|, \qquad x \in (-\infty, -1) \text{ nebo } x \in (-1,\infty),\\
	I_2 &= \int \frac{x+1}{x^2+x+1}\d x = \frac{1}{2} \underbrace{\int \frac{2x+1}{x^2+x+1}\d x}_{I_3} + \frac{1}{2} \underbrace{\int \frac{1}{x^2+x+1}\d x }_{I_4},
\end{align*}
je tedy
\begin{align*}
I_3 &\stackrel{c}{=} \log (x^2+x+1), \qquad x \in \mathbb R, \\
I_4 &= \int \frac{1}{\left( x+\frac{1}{2} \right )^2 + \frac{3}{4} } \d x= \frac{4}{3} \int \frac{1}{\left( \frac{2x+1}{\sqrt{3}} \right )^2 +1} \d x \\
&\stackrel{c}{=} \frac{4}{3}\cdot \frac{\sqrt{3}}{2} \arctg{\frac{2x+1}{\sqrt{3}}}, \qquad x \in \mathbb R.
\end{align*}
Původní integrál je tedy roven
$$
    \int\frac{x^2+1}{(x+1)(x^2+x+1)}\d x \stackrel{c}{=} 2\log |x+1| - \frac{1}{2} \log (x^2+x+1) - \frac{1}{\sqrt{3}}\arctg \frac{2x+1}{\sqrt{3}},
$$
kde $x \in (-\infty, -1)$ nebo $x \in (-1,\infty)$.
\end{priklad}

\begin{definice}
\textbf{Polynomem dvou proměnných} rozumíme funkci
\[
[u,v] \mapsto \sum_{i,j=0}^n a_{ij} u^i v^j,
\]
kde $n \in \mathbb{N}\cup\{0\}$, $a_{ij} \in \mathbb{R}$ pro $i,j\in \left \{ 0,\dots,n \right \} $.
\textbf{Racionální funkcí} dvou proměnných rozumíme podíl polynomů dvou proměnných, kde polynom ve jmenovateli není identicky roven nule.
\end{definice}

\begin{definice}
Řekneme, že $R$ je \textbf{lichá v první proměnné}, pokud pro každé $(x,y)\in \mathcal D(R)$
platí $(-x,y)\in \mathcal D(R)$ a $R(-x,y) = -R(x,y)$. Analogicky definujeme \textbf{lichost ve druhé proměnné}.
Funkce $R$ je \textbf{sudá}, pokud pro každé $(-x,-y)\in \mathcal D(R)$ platí $(-x,-y)\in \mathcal D(R)$ a $R(x,y) = R(-x,-y)$.
\end{definice}

\paragraph{Goniometrické substituce.} Nechť $R$ je racionální funkce dvou proměnných a $I$ otevřený neprázdný interval.
Uvažujme integrál tvaru
\[
\int R(\sin t, \cos t)\d t, \qquad t \in I,
\]
přičemž integrand je definován na intervalu $I$. Pro převedení úlohy na integraci racionální funkce
lze užít následujících substitucí.
\begin{enumerate}[(a)]
	\item Je-li $R$ lichá ve druhé proměnné, lze užít substituci $\sin t = x$.
\item Je-li $R$ lichá v první proměnné, lze užít substituci $\cos t = x$.
\item Je-li $R$ sudá, lze užít susbstituci $\tg t = x$.
\item Vždy lze použít $\tg(t/2) = x$.
\end{enumerate}

\begin{priklad}
	Spočtěte $\int \frac{1}{1+\sin^2t}\d t = \int g(t) \d t$.
	\par Máme $R(a,b)=\frac{1}{1+a^2}$. Tato funkce je sudá, použijeme tedy substituci $\tg t = x$. Bude tedy
	$$
	    \varphi(t)=\tg t, \qquad t \in \left( -\frac{\pi}{2}, \frac{\pi}{2} \right )+k\pi.
	$$
	Pak je
	$$
	    \varphi^\prime(t) = \frac{1}{\cos^2 t}, \qquad t \in \left( -\frac{\pi}{2}, \frac{\pi}{2} \right )+k\pi.
	$$
	Chceme $\tg t = x$. Dostaneme
	\begin{align*}
	\frac{\sin^2 t}{\cos^2 t} &= x^2\\
	\sin^2 t &= x^2 \cos ^2 t \\
	\sin^2 t &= x^2 (1-\sin^2 t)\\
	\sin^2 t &= \frac{x^2}{1+x^2},\\
	\cos^2 t &= \frac{1}{1+x^2}.
	\end{align*}
	Přepíšeme
	$$
	\int g(t) \d t = \int \frac{1}{1+\sin^2 t} \cos^2 t \underbrace{\frac{1}{\cos^2 t}}_{\varphi(t)}\d t,
	$$
	pak je
	$$
	f(x) = \frac{\frac{1}{1+x^3}}{1+\frac{x^2}{1+x^2}},
	$$
	a tedy
	$$
	    \int f(x) \d x = \int \frac{1}{1+2x^2} \d x = \int \frac{1}{1+(\sqrt{2}x)^2} \stackrel{c}{=} \frac{1}{\sqrt{2}}\arctg{(\sqrt{2}x)},
	$$
	původní integrál je tedy roven
	$$
	    \int \frac{1}{1+\sin^2 t} \d t \stackrel{c}{=} \frac{1}{\sqrt{2}} \arctg{(\sqrt{2}\tg{t})}, \qquad t \in \left( -\frac{\pi}{2}, \frac{\pi}{2} \right )+k\pi, \pi \in \mathbb Z .
	$$
	Protože je primitivní funkce spojitá, ale tato funkce má body nespojitosti, musíme jednotlivé
	hodnoty \uv{posunout} tak, aby limita zleva se rovnala limitě zprava v bodech nespojitosti.
	Nakonec musíme ověřit, že derivace v bodech nespojitosti opravdu vyjde správně.
\end{priklad}

\paragraph{Několik dalších substitucí.} Ať $R$ je racionální funkce dvou proměnných.
\begin{itemize}
	\item $\int R \left( t, \sqrt[q]{\frac{az+b}{ct+f}} \right ) \d t$, kde $q \in \mathbb N , a,b,c,f,\in \mathbb R , af \ne bc$
	Položíme
	$$
	    x = \sqrt[q]{\frac{at+b}{ct+f}}.
	$$
	\item $\int R \left( t, \sqrt{at^2 + bt +c} \right )\d t $, kde $a,b,c \in \mathbb R , a \ne 0$.
	Rozlišíme následující možnosti:
	\begin{enumerate}[(1)]
		\item Polynom $at^2+bt+c$ má dvojnásobný kořen $\alpha$. Pak můžeme psát
		$$
		    at^2+bt+c = a(t-\alpha)^2,
		$$
		a tedy
		$$
  \sqrt{at^2+bt+c}= \sqrt{a} |t-\alpha|, \qquad a > 0.
		$$
		\item Polynom $at^2+bt+c$ má dva různé reálné kořeny $\alpha_1, \alpha_2$, $\alpha_1 <\alpha_2$.
		Potom můžeme psát
		$$
  \sqrt{at^2+bt+c} = |t-\alpha_1| \cdot \sqrt{\frac{a(t-\alpha_2)}{t-\alpha_1}}, \qquad a>0.
		$$
		\item Polynom $at^2+bt+c$ nemá reálné kořeny. Potom $a>0$, a tedy i $c>0$, aby vůbec
		úloha měla dobrý smysl. V tom případě můžeme použít tzv. Eulerovy substituce:
		\begin{align*}
			\sqrt{at^2+bt+c} &= \sqrt{a}t +x, \\
			\sqrt{at^2+bt+c} &= -\sqrt{a}t+x, \\
			\sqrt{at^2+bt+c} &= tx+ \sqrt{c}.
		\end{align*}
		Tyto substituce vedou na integraci racionální funkce.
	\end{enumerate}
\end{itemize}

\begin{priklad}
	Spočtěte $\int \frac{1}{t^2 \sqrt{t^2+1}}\d t$.
	\par Budeme zvlášť uvažovat intervaly $(-\infty, 0), (0, \infty)$. Za sbustituci zvolíme
	$\sqrt{t^2+1}=t+x$. Pak je
	$$
	    \varphi(t) = \sqrt{t^2+1}-t,
	$$
	$\varphi: \mathbb R \to \mathbb R $, a
	$$
	    \varphi^\prime(t) = \frac{1}{2}\cdot \frac{2t}{\sqrt{t^2+1}} - 1 = \frac{t^2}{t^2+1}-1.
	$$
	Máme tedy
	\begin{align*}
		t^2+t &= t^2+2tx+ x^2 \\
		t &= 2tx +x^2\\
		t &= \frac{1-x^2}{2x},\\
		\d t &= \frac{-2x\cdot 2x - (1-x^2)\cdot 2}{4x^2}\d x = \frac{-2-2x^2}{4x^2} \d x.
	\end{align*}
	Můžeme tedy psát
	\begin{align*}
	&\int \frac{1}{\left( \frac{1-x^2}{2x} \right )^2 \left( \frac{1-x^2}{2x}+x \right )  } \cdot \frac{-2-2x^2}{4x^2}\d x\\
	=&\int \frac{-4x}{(1-x)^2(1+x)^2}\d x \\
	=& \int \left( \frac{-1}{(1-x)^2}+\frac{1}{(1+x)^2} \right )\d x \\
	\stackrel{c}{=}& -\frac{1}{1-x} - \frac{1}{1+x} = \frac{2}{x^2-1},
	\end{align*}
	kde $x \in (-\infty, -1)$, $x \in (-1,1)$ nebo $x\in (1,\infty)$.
	Celkem tedy
	$$
	    \int \frac{1}{t^2 \sqrt{t^2+1}}\d t \stackrel{c}{=} \frac{1}{\left( \sqrt{t^2+1}t \right )^2 - 1 },
	$$
	kde $t \in (-\infty, 0)$, $t \in (0, \infty)$.
\end{priklad}

\subsection{Riemannův integrál}
\begin{definice}
Konečnou posloupnost $\{x_j\}_{j=0}^n$ nazýváme \textbf{dělením intervalu} $[a,b]$, jestliže
\[
a = x_0 < x_1 < \cdots < x_n = b.
\]
Body $x_0,\dots,x_n$ nazýváme \textbf{dělícími body}. \textbf{Normou dělení} $D=\left \{ x_j \right \}_{j=0}^n $
rozumíme číslo
\[
\nu(D) = \max \{x_j - x_{j-1}; j\in \left \{ 1,\dots,n \right \} \}.
\]
Řekneme, že dělení $D^\prime$ intervalu $[a,b]$ je \textbf{zjemněním dělení} $D$ intervalu $[a,b]$,
jestliže každý dělící bod $D$ je i dělícím bodem $D^\prime$.
\end{definice}

\begin{definice}
Nechť $f$ je omezená funkce definovaná na intervalu $[a,b]$ a $D=\{x_j\}_{j=0}^n$ je dělení $[a,b]$.
Označme
\begin{align*}
	\overline{S}(f,D) &= \sum_{j=1}^n M_j (x_j - x_{j-1}), \quad\text{kde}\ M_j = \sup \left \{ f(x);x \in [x_{j-1},x_j] \right \} ,\\
	\underline S(f,D) &= \sum_{j=1}^n m_j (x_j - x_{j-1}), \quad\text{kde}\ m_j = \inf \left \{ f(x);x \in [x_{j-1},x_j] \right \},
\end{align*}
\begin{align*}
\overline{\int_a^b} f(x)\d x &= \inf \left \{ \overline S(f,D);D\ \text{je dělením intervalu}\ [a,b] \right \}, \\
\underline{\int_a^b} f(x)\d x &= \sup \left \{ \underline S(f,D);D\ \text{je dělením intervalu}\ [a,b] \right \}.
\end{align*}
\end{definice}

\begin{definice}\,
\begin{itemize}
	\item Řekneme, že omezená funkce $f$ na intervalu $[a,b]$, $a<b$, má \textbf{Riemannův integrál od $a$ do $b$},
pokud $\overline{\int_a^b} f(x) \d x = \underline{\int_a^b} f(x) \d x$. Hodnota integrálu $f$ od $a$ do $b$
je rovna této společné hodnotě. Značíme ji $\int_a^b f(x) \d x$.
\item Pokud $a>b$, definujeme $\int_a^b f(x) \d x = -\int_b^a f(x) \d x$. V případě, že $a=b$, definujeme
$\int_a^b f(x) \d x = 0$.
\end{itemize}
\end{definice}

\begin{definice}
Nechť $a,b \in \mathbb R $. Množinou všech funkcí, které mají Riemannův integrál od $a$ do $b$,
značíme $\mathcal R([a,b])$.
\end{definice}

\begin{poznamka}
	\,
	\begin{enumerate}[(1)]
		\item Platí $\int_0^1 1 \d x = 1$.
		\item Funkce $\int_0^1 D(x) \d x$, kde
		$$
		    D(x) = \begin{cases}
						0, & x \in \mathbb R \smallsetminus \mathbb Q,\\
						1, & x \in \mathbb Q
					\end{cases}
		$$
		je Dirichletova funkce, není riemannovsky integrovatelná. Vezměme $\overline{S}(D,P)$, kde
		$P$ je dělení intervalu $[0,1]$. Pak je
		\begin{align*}
		\overline{S}(D,P) &= \sum _{j=1}^n M_j(x_j-x_{j-1})=1,\\
		\underline{S}(D,P) &= \sum_{j=1}^n m_j(x_j-x_{j-1}) = 0.
		\end{align*}
		Máme tedy, že
		$$
		    \overline{\int_0^1} D(x)\d x = 1, \qquad \underline{\int_0^1} D(x) \d x = 0,
		$$
		a tedy $D(x)$ není riemannovsky integrovatelná.
	\end{enumerate}
\end{poznamka}

\begin{lemma} \label{lemma8.10}
Nechť $f$ je omezená funkce na intervalu $[a,b]$.
\begin{enumerate}[(a)]
	\item Nechť $D, D^\prime$ jsou dělení $[a,b]$ a $D^\prime$ zjemňuje $D$. Pak platí
\[
\underline{S}(f,D) \le \underline{S}(f,D^\prime) \le \overline{S}(f,D^\prime) \le \overline{S}(f,D).
\]
\item Nechť $D_1, D_2$ jsou dělení intervalu $[a,b]$. Pak platí
\[
\underline{S}(f,D_1) \le \overline{S}(f,D_2).
\]
\item Platí
\[
\underline{\int_a^b} f(x)\d x \le \overline{\int_a^b} f(x)\d x.
\]
\end{enumerate}
\end{lemma}

\begin{proof}
	\,
	\begin{enumerate}[(a)]
		\item Ať je $D = \left \{ x_j \right \}_{j=0}^n $. BÚNO $D^\prime$ obsahuje dělící body
		$D$ a navíc právě jeden bod $z \in \left( x_{j-1}, x_j \right ) $, $j \in \left \{ 1,\dots, n \right \} $.
		Můžeme psát
		\begin{align*}
		    \underline{S}(f,D) &= \sum_{k=1, k \ne j}^n m_k (x_k-x_{k-1})+ m_j(x_j-x_{j-1}),\\
			\underline{S}(f,D) &= \sum_{k=1, k \ne j}^n m_k (x_k-x_{k-1})+ \underbrace{m_j^\prime}_{\geq m_j}(z-x_{j-1}) + \underbrace{m^{\prime\prime}_j}_{\geq m_j} (x_j - z).
		\end{align*}
		Tím jsme dokázali první nerovnost. Druhá nerovnost je triviální a třetí nerovnost dokážeme analogicky.
		\item Nechť dělení $D$ zjemňuje $D_1$ i $D_2$. Podle části (a) platí
		$$
		    \underline{S}(f,D_1) \leq \underline{S}(f,D) \leq \overline{S}(f,D) \leq \overline{S}(f,D_2).
		$$
		\item Zvolme dělení $D_1$ intervalu $[a,b]$. Potom pro každé dělení $D_2$ platí
		$$
		    \underline{S}(f,D_1) \leq  \overline{S}(f,D_2).
		$$
		Odtud máme, že
		$$\underline{S}(f,D_1) \leq \overline{\int_a^b}f=\inf \left \{ \overline{S}(f,D_2); D_2 \text{ dělení } [a,b]\right \}. $$
		Jinými slovy,
		$$
		    \underline{S}(f,D_1) \leq \overline{\int_a^b}f
		$$
		pro každé dělení $D_1$. Je
		$$
		    \underline{\int_a^b } f = \sup \left \{ \underline{S}(f,D_1); D_1 \text{ je dělením }[a,b] \right \},
		$$
		odkud máme
		$$
		    \underline{\int_a^b}f \leq \overline{\int_a^b}f.
		$$
		\qedhere
	\end{enumerate}
\end{proof}

\begin{dusledek}\label{dusledek8.11}
Nechť $f$ je omezená na $[a,b]$, $D_1$ a $D_2$ jsou dělení intervalu $[a,b]$. Potom
\[
m(b-a) \le \underline{S}(f,D_1) \le \underline{\int_a^b} f(x)\d x
\le \overline{\int_a^b} f(x)\d x \le \overline{S}(f,D_2) \le M(b-a),
\]
kde $m = \inf\{f(x);\, x \in [a,b]\}$ a $M = \sup\{f(x);\, x \in [a,b]\}$.
\end{dusledek}

\begin{proof}
	Zvolme dělení $D$ sestávající z bodů $a$ a $b$. Potom platí
	$$
	    m(b-a) = \underline{S}(f,D)\leq \underline{S}(f,D_1) \leq \underline{\int_a^b}f \leq \overline{\int_a^b}f \leq \overline{S}(f,D_2) \leq \overline{S}(f,D)=M(b-a).
	$$
\end{proof}

\begin{veta}\label{veta8.12}
Nechť $f$ je omezená na $[a,b]$. Pak pro každé $\varepsilon > 0$ existuje $\delta > 0$ takové, že pro každé dělení $D$ intervalu $[a,b]$ splňující $\nu(D) < \delta$ platí
\begin{align*}
	&\underline{\int_a^b} f(x)\d x \ge \underline{S}(f,D) \ge \underline{\int_a^b} f(x)\d x - \varepsilon,\\
&	\overline{\int_a^b} f(x)\,dx \le \overline{S}(f,D) \le \overline{\int_a^b} f(x)\,dx + \varepsilon.
\end{align*}
\end{veta}

\begin{proof}
    Nalezneme  $K>0$ takové, že pro každé $x \in [a,b]$ platí $|f(x)|<K$. Nyní zvolme $\varepsilon >0$.
    Nalezneme dělení $D_0 = \left \{ x_j \right \}_{j=0}^n $ intervalu $[a,b]$ takové, že
    $$
        \overline{S}(f,D_0) < \overline{\int_a^b}f+ \frac{\varepsilon}{2}.
    $$
    Nalezneme $\delta >0$ splňující:
    \begin{enumerate}[(1)]
	\item $\delta < \min \left \{ x_j-x_{j-1}; j \in \left \{ 1,\dots,n \right \}  \right \} $,
	\item $\delta < \frac{\varepsilon}{8Kn}$.
    \end{enumerate}
    Zvolme dělení $D$ na intervalu $[a,b]$ splňující $\nu(D)<\delta$. Chceme ukázat
    $$
        \overline{\int_a^b}f \leq \overline {S}(f,D) < \overline{\int_a^b}f+\varepsilon.
    $$
    Dělení $P$ bude obsahovat právě ty body dělení $D_0$ a $D$. Označme $\mathscr D$ (resp. $\mathscr P$) jako intervaly
    příslušné dělení $D$ (resp. dělení $P$). Počítáme
    $$
        0\leq\overline{S}(f,D) - \overline{S}(f,P) = \sum_{I\in \mathscr D}\sup_I f \cdot \text{délka}(I) - \sum_{j\in \mathscr P}\sup_J f \cdot \text{délka}(J).
    $$
    Dále označme $Z = \left \{ x_0,x_1,\dots,x_n \right \} $. Pak je
    \begin{align*}
    \overline{S}(f,D) - \overline{S}(f,P) &= \sum_{\substack{I\in \mathscr D \\ I\cap Z \ne \emptyset}} \sup_I f \cdot \text{délka}(I)-\sum_{\substack{J\in \mathscr P\\ J\cap Z \ne \emptyset}} \sup_j f \cdot \text{délka}(J)\\
    &\leq 2n\cdot 2K\cdot \delta = 4Kn\delta<\frac{\varepsilon}{2}.
    \end{align*}
    Dokázali jsme:
    $$
        \overline{\int _a^b}f\leq \overline{S}(f,P) \leq \overline{S}(f,D_0) \leq \overline{\int_a^b}f+\frac{\varepsilon}{2}
    $$
    a taky
    $$
        \overline{\int_a^b}f \leq \overline{S}(f,D) \leq \overline S(f,P)+\frac{\varepsilon}{2} < \overline{\int_a^b} f + \frac{\varepsilon}{2}+\frac{\varepsilon}{2}=\overline{\int_a^b}f+\varepsilon.
    $$
    Analogicky i pro $\underline{\int_a^b}f$.
\end{proof}

\begin{dusledek}\label{dusledek8.13}
Nechť $a,b \in \mathbb{R}$, $a<b$, a $f$ je omezená funkce na $[a,b]$. Nechť $\{D_n\}_{n=1}^\infty$ je posloupnost dělení intervalu $[a,b]$ splňující $\lim \nu(D_n) = 0$. Pak
\[
\underline{\int_a^b} f(x)\d x = \lim_{n\to\infty} \underline{S}(f,D_n)
\qquad \text{a} \qquad
\overline{\int_a^b} f(x)\d x = \lim_{n\to\infty} \overline{S}(f,D_n).
\]
\end{dusledek}

\begin{proof}
	Zvolme $\varepsilon > 0$. Podle věty \ref{veta8.12} nalezneme $\delta >0$ takové, že pro každé
	dělení $D$, $\nu(D)<\delta$ platí
	$$
	    \overline{\int_a^b}f \leq \overline{S}(f,D)<\overline{\int_a^b}f + \varepsilon.
	$$
	Nalezneme $n_0 \in \mathbb N $ takové, že
	$$
	    \forall n \geq n_0: \nu(D_n)<\delta.
	$$
	Potom
	$$
	    \forall n \geq n_0: \overline{\int_a^b}f \leq \overline{S}(f,D_n) < \overline{\int_a^b}f+\varepsilon,
	$$
	a tedy
	$$
 \lim_{n \to \infty} \overline{S}(f,D_n) = \int_a^b f.
	$$
\end{proof}

\begin{priklad}
	Spočtěte $\int_0^1 x^2\d x$.
	\par Položme $D_n = \left \{ \frac{j}{n} \right \}_{j=0}^1 $ a označme $f(x)=x^2$.
		$$
		\overline{S}(f,D_n) = \sum_{j=1}^n \left( \frac{j}{n} \right )^2 \cdot \frac{1}{n} = \frac{1}{n^3} \sum_{j=1}^n j^2 = \frac{1}{n^2} \cdot \frac{n(n+1)(2n+1)}{6} \stackrel{n\to \infty}{\longrightarrow} \frac{1}{3}.
		$$
		Proto je
		$$
		    \overline{\int_0^1}  x^2 \d x = \frac{1}{3}.
		$$
		Obdobně
		$$
		\underline{S}(f,D_n) = \sum_{j=1}^n \left( \frac{j-1}{n} \right )^2 \cdot \frac{1}{n} = \frac{1}{n^3}\sum_{j=1}^n (j-1)^2 = \frac{1}{n^3}\cdot \frac{(n-1)n(2n-1)}{6}\stackrel{n\to \infty}{\longrightarrow} \frac{1}{3} ,
		$$
		a tedy
		$$
		    \underline{\int _0^1}x^2 \d x = \frac{1}{3},
		$$
		a tedy i
		$$
		    \int_0^1 x^2 \d x = \frac{1}{3}.
		$$
\end{priklad}

\begin{veta}[Kritérium existence Riemannova integrálu]\label{veta8.14}
Nechť $a,b \in \mathbb{R}$, $a<b$, a $f$ je omezená funkce na $[a,b]$. Pak jsou následující výroky ekvivalentní:
\begin{enumerate}[(i)]
	\item $f \in \mathcal R([a,b])$,
	\item pro každé $\varepsilon > 0$ existuje dělení $D$ intervalu $[a,b]$ takové, že
\[
\overline{S}(f,D) - \underline{S}(f,D) < \varepsilon.
\]
\end{enumerate}
\end{veta}

\begin{proof}
	\,
	\begin{enumerate}
		\item[$\implies$] Z $f \in \mathcal R(\left [ a,b \right ] )$ platí
		$$
		    \underline{\int_a^b} f = \overline{\int_a^b} f.
		$$
		Zvolme $\varepsilon > 0$. Nalezneme dělení $D_1$ intervalu $[a,b]$ splňující
		$$
		    \int_a^b f \leq \overline{S}(f,D_1)< \int_a^b f + \frac{\varepsilon}{2}.
		$$
		Dále nalezneme dělení $D_2$ intervalu $[a,b]$ splňující
		$$
		    \int_a^b f - \frac{\varepsilon}{2}\underline{S}(f,D_2) \leq \int_a^b f.
		$$
		Zvolme dělení $D$ jako zjemnění $D_1$ a $D_2$. Potom platí
		$$
		    \overline{S}(f,D)-\underline{S}(f,D) \leq \overline{S}(f,D_1 - \underline S(f,D_2) < \int_a^b f + \frac{\varepsilon}{2} - \left( \int_a^b f - \frac{\varepsilon}{2} \right )= \varepsilon.
		$$
		\item[$\impliedby$] Zvolme $\varepsilon>0$. K němu nalezneme podle (ii) dělení
		$D$ intervalu $[a,b]$ splňující
		$$
		    \overline S(f,D) - \underline S(f,D) < \varepsilon.
		$$
		Podle lemmatu \ref{lemma8.10}(c) je
		$$
		    0 \leq \overline{\int_a^b} f - \underline{\int_a^b} f \leq \overline S(f,D) - \underline S(f,D) < \varepsilon.
		$$
		Odtud máme $\overline{\int_a^b}f=\underline{\int_a^b}f$.\qedhere
	\end{enumerate}
\end{proof}

\begin{definice}
Řekneme, že funkce $f$ je \textbf{stejnoměrně spojitá} na intervalu $I$, jestliže platí
\[
\forall \varepsilon > 0\, \exists \delta > 0\, \forall x,y \in I: (|x-y| < \delta \implies |f(x)-f(y)| < \varepsilon).
\]
\end{definice}

\begin{poznamka}[Spojitost versus stejnoměrná spojitost]\,
\begin{enumerate}[(1)]
	\item Funkce $f$ je spojitá na intervalu $I$ pokud
	$$
	\forall x \in I \, \forall \varepsilon> 0 \, \forall \delta >0 \, \forall y \in I: \left( |x-y| < \delta \implies |f(x)-f(y)|<\varepsilon \right ).
	$$
	Funkce $f$ je stejnoměrné spojitá na intervalu $I$, pokud
	$$
	    \forall  \varepsilon > 0 \, \exists \delta > 0 \, \forall x \in I\, \forall y \in I: \left( |x-y| < \delta \implies |f(x)-f(y)|<\varepsilon \right ).
	$$
	Odsud máme, že
	$$
	    f \text{ je stejnoměrně spojitá} \implies f \text{ je spojitá}.
	$$
	Opačná implikace obecně neplatí.
	\item Vezměme $I=(0,1)$, $f(x)= \frac{1}{x}, x \in I$. Funkce $f$ je na $I$ spojitá. Bodem $x_n$ značíme
	bod $\frac{1}{n}$.
	\par Chceme ukázat, že $f$ není stejnoměrně spojitá na $I$. Položme $\varepsilon = 1$.
	Nechť $\delta >0$. Nalezneme $n\in \mathbb N $ takové, že $x_n \in (0,\delta) \cap I$. Potom
	$x_{n+1}\cap I$ a $|x_n - x_{n+1}|<\delta$. Pak je
	$$
	    f(x_{n+1})-f(x_n) = \frac{1}{\frac{1}{n+1}} - \frac{1}{\frac{1}{n}} = n+1-n = 1,
	$$
	a tedy
	$$
	    |f(x_{n+1})-f(x_n)| = 1.
	$$
	Proto $f$ není stejnoměrně spojitá.
\end{enumerate}
\end{poznamka}

\begin{veta}\label{veta8.15}
Nechť $a,b \in \mathbb{R}$, $a<b$, a $f$ je spojitá funkce na $[a,b]$. Potom $f$ je stejnoměrně spojitá na $[a,b]$.
\end{veta}

\begin{proof}
	Pro spor předpokládejme, že
	\begin{equation}\label{vlastnostss}
	\exists \varepsilon > 0 \, \forall \delta > 0 \, \exists x \in I \, \exists y \in I : \left( |x-y| < \delta\implies |f(x)-f(y)|<\varepsilon \right ).
	\end{equation}
	Nechť $\varepsilon>0$ splňuje (\ref{vlastnostss}). Odsud pro každé $n \in \mathbb N $ nalezneme
	dvojici $x_n, y_n \in I$ splňující
	$$
	    |x_n - y_n| < \frac{1}{n} \land |f(x_n)-f(x_n)|\geq \varepsilon.
	$$
	Podle Weierstrassovy věty \ref{bolzano-weierstrass} nalezneme konvergentní podposloupnost
	$\left \{ x_{n_k} \right \}_{k=1}^\infty $ posloupnosti $\left \{ x_n \right \}_{n=1}^\infty $
	splňující
	$$
 \lim_{k \to \infty} x_{n_k} = x^*.
	$$
	Platí $x^* \in I$ a dále pro každé $k \in \mathbb N $ máme
	$$
	    0\leq|x^* - y_{n_k}| \leq \underbrace{|x^*-x_{n_k}|}_{\to \, 0}+\underbrace{|x_{n_k}-y_{n_k}|}_{\to\, 0}.
	$$
	Odsud máme, že $\lim_{k \to \infty} y_{n_k}=x^*$.
	Funkce $f$ je spojitá v $x^*$ vzhledem k $I$, tj.
	\begin{equation}\label{podminka}
	\exists \delta > 0 \, \forall y \in I: |x^* - y|<\delta \implies |f(x^*)-f(y)|<\frac{\varepsilon}{2}.
\end{equation}
	Nechť $\delta >0$ splňuje (\ref{podminka}). K němu nalezneme $k \in \mathbb N $ takové, že
	$$
	    |x_{n_k}-x^*|< \delta \land |y_{n_k} - x^*| <\delta.
	$$
	Potom platí
	$$
	    \varepsilon \leq|f(x_{n_k}) - f(y_{n_k})|\leq |f(x_{n_k})-f(x^*)| + |f(x^*)-f(y_{n_k})| < \frac{\varepsilon }{2}+\frac{\varepsilon}{2}= \varepsilon,
	$$
	odkud máme spor.
\end{proof}

\begin{veta}\label{veta8.16}
Nechť $a,b \in \mathbb{R}$, $a<b$, a $f$ je spojitá funkce na $[a,b]$. Potom $f \in \mathcal R([a,b])$.
\end{veta}

\begin{proof}
	Zvolme $\varepsilon > 0$. Podle věty \ref{veta8.15} existuje $\delta > 0$ takové, že
	$$
	    \forall x ,y \in [a,b], |x-y| < \delta \implies |f(x)-f(y)| < \varepsilon.
	$$
	Nalezneme dělení $D$ intervalu $[a,b]$ takové, že
	$D = \left \{ x_i \right \}_{i=0}^n $ a $\nu (D)<\delta$. Označme
	$$
	m_i = \inf_{[x_{i-1},x_i]} f, \qquad M_i = \sup_{[x_{i-1},x_i]} f.
	$$
	Z (1) a (2) plyne $m_i \leq M_i \leq m_i+\varepsilon$. Počítejme
	\begin{align*}
	0 &\leq \overline S(f,D) - \underline S(f,D) = \sum_{i=1}^n M_i(x_i-x_{i-1}) - \sum_{i=1}^n m_i(x_i - x_{i-1}) \\
	&= \sum_{i=1}^n \underbrace{(M_i-m_i)}_{\leq\, \varepsilon}(x_i-x_{i-1}) \leq \sum_{i=1}^n \varepsilon (x_i-x_{i-1}) = \varepsilon (b-a).
	\end{align*}
	Právě jsme dokázali (ii) ve větě \ref{veta8.14}, a tedy $f\in \mathcal R([a,b])$.
\end{proof}

\begin{veta}\label{veta8.17}
Nechť $a,b \in \mathbb{R}$, $a<b$, a $f$ je monotónní funkce na $[a,b]$. Potom $f \in \mathcal R([a,b])$.
\end{veta}

\begin{proof}
	BÚNO  $f$ je neklesající. Pro každé $x \in \left [ a,b \right ] $ platí $f(a)\leq f(x)\leq f(b)$.
	Funkce $f$ je tedy omezená. Nalezneme $K > 0$ takové, že
	$$
	    \forall x \in \left [ a,b \right ] : |f(x)|\leq K.
	$$
	Zvolme $\varepsilon >0$. K němu nalezneme dostatečně velké $n \in \mathbb N $. Položme
	$$
	    x_j = a+ \frac{b-a}{n}\cdot j, \qquad j=0,\dots,n,
	$$
	$\mathcal D = \left \{ x_j \right \}_{j=0}^n. $ Počítejme
	\begin{align*}
	\overline S(f,D) &= \sum_{j=1}^n \sup_{[x_{j-1},x_j]}f \cdot (x_j-x_{j-1}) = \sum _{j=1}^n f(x_j)\cdot \frac{b-a}{n},\\
	\underline S(f,D) &= \sum_{j=1}^n \inf{[x_{j-1},x_j]}f \cdot (x_j-x_{j-1}) = \sum _{j=1}^n f(x_{j-1})\cdot \frac{b-a}{n}.
	\end{align*}
	Je tedy
	\begin{align*}
	\overline S(f,D) - \underline S(f,D) &= \sum_{j=1}^n \left( f(x_j)-f(x_{j-1}) \right )\cdot \frac{b-a}{n} \\
	&= \frac{b-a}{n} \sum_{j=1}^n \left( f(x_j)-f(x_{j-1}) \right ) \\
	&= \frac{b-a}{n} \left( f(b) -f(a)\right ) <\varepsilon,
\end{align*}
pokud $n > \frac{1}{\varepsilon}(b-a)(f(b)-f(a))$. Odtud vyplývá podle věty \ref{veta8.14}, že $f \in \mathcal R([a,b])$.
\end{proof}

\begin{veta}[Linearita Riemannova integrálu]\label{veta8.18}
Nechť $a,b \in \mathbb{R}$, $a<b$, $f,g \in \mathcal R([a,b])$ a $\alpha \in \mathbb{R}$. Pak $f+g \in \mathcal R([a,b])$, $\alpha f \in \mathcal R([a,b])$ a platí
\begin{align*}
\int_a^b (f(x) + g(x))\d x &= \int_a^b f(x)\,dx + \int_a^b g(x)\d x,\\
\int_a^b \alpha f(x)\d x &= \alpha \int_a^b f(x)\d x.
\end{align*}
\end{veta}

\begin{proof}
	Protože jsou funkce $f,g$ omezené, je i $f+g$ omezená. Nechť $I\subseteq [a,b]$ je neprázdný interval.
	Potom platí
	$$
	\sup_I (f+g) \leq \sup_I f+ \sup_I g.
	$$
	(Je totiž pro $x\in I$: $f(x)+g(x)\leq \sup_I f + \sup_I g$.) Podobně pro infimum platí
	$$
	    \inf_I f + \inf_I g \leq \inf_I (f+g).
	$$
	Pro libovolné dělení $\mathcal D$ intervalu $[a,b]$ platí
	$$
	    \underline S(f,D)+\underline S(g,D)\leq\underline S(f+g,D) \leq \overline S(f+g,D)\leq \overline S(f,D)+\overline S(g,D).
	$$
	Nalezneme posloupnost dělení $\left \{ \mathcal D^n \right \}_{n=1}^\infty $ intervalu $[a,b]$ splňující
	$\lim_{n\to \infty} \nu (\mathcal D^n)=0$. Potom platí podle důsledku \ref{dusledek8.13}
	\begin{align*}
        \lim_{n \to \infty} \overline S(f,D^n)&=\lim_{n \to \infty} \underline S(f,D^n)=\int_a^b f,\\
		\lim_{n \to \infty} \overline S(g,D^n)&=\lim_{n \to \infty} \underline S(g,D^n)=\int_a^b g.
	\end{align*}
	Pro každé $n\in \mathbb N $ platí
	$$
	    \underbrace{\underline S(f,D^n) + \underline S(g,D^n)}_{\to \int_a^b f +\int_a^b g}\leq \underline S(f+g,D^n) \leq \overline S(f+g,D^n)\leq \underbrace{\overline S(f,D^n)+ \overline S(g,D^n)}_{\to \int_a^b f + \int_a^b g},
	$$
	a tedy podle věty o dvou strážnících máme
	$$
 \lim_{n \to \infty} \underline S(f+g,D^n) = \lim_{n \to \infty} \overline S(f+g,D^n)=\int_a^b f + \int_a^b g.
	$$
	Podle důsledku \ref{dusledek8.13} dostáváme, že $f+g\in \mathcal R([a,b])$ a platí
	$$
	    \int_a^b (f+g) = \int_a^bf + \int _a^b g.
	$$
	Pro $\alpha$ a neprázdný interval $I\subseteq [a,b]$ platí
	$$
	    \sup_I (\alpha f) = \begin{cases}
						\alpha \sup _I f, & \text{pro} \ \alpha \geq 0,\\
						\alpha \inf _I f, & \text{pro} \ \alpha < 0.
					\end{cases}
	$$
	Obdobně pro infimum
	$$
	    \inf_I (\alpha f) = \begin{cases}
					\alpha \inf _I f, & \text{pro} \ \alpha \geq 0,\\
					\alpha \sup _I f, & \text{pro} \ \alpha < 0.
					\end{cases}
	$$
	Potom platí
	$$
 \lim_{n \to \infty} \overline S(\alpha f, D^n) = \left .\begin{cases}
 \lim_{n \to \infty} \alpha \overline S(f,D^n), & \text{pro} \ \alpha \geq 0, \\
 \lim_{n \to \infty} \alpha \underline S(f,D^n) & \text{pro} \ \alpha < 0.
 \end{cases} \right \} = \alpha \int_a^b f,
	$$
	podobně
	$$
	\lim_{n \to \infty} \underline S(\alpha f, D^n) =\left . \begin{cases}
 \lim_{n \to \infty} \alpha \underline S(f,D^n), & \text{pro} \ \alpha \geq 0, \\
 \lim_{n \to \infty} \alpha \overline S(f,D^n) & \text{pro} \ \alpha < 0.
 \end{cases}\right \} =  \alpha \int_a^b f,
	$$
	odkud podle důsledku \ref{dusledek8.13} dostáváme, že $f\in \mathcal R([a,b])$ a
	$\int_a^b \alpha f  = \alpha \int _a^b f$.
\end{proof}

\begin{veta}[Riemannův integrál a uspořádání]\label{veta8.19}
Nechť $a,b \in \mathbb{R}$, $a<b$, $f,g \in \mathcal R([a,b])$ a $f(x) \le g(x)$ pro každé $x \in [a,b]$. Pak platí
\[
\int_a^b f(x)\d x \le \int_a^b g(x)\d x.
\]
\end{veta}

\begin{proof}
	Nechť $\left \{ D^n \right \} $ je jako v předchozím důkazu. Potom platí pro každé $n \in \mathbb N $
	$$
	    \underbrace{\overline S(f, D^n)}_{\to \int _a^b f} \leq \underbrace{\overline S(g,D^n)}_{\to \int_a^b g}.
	$$
	(Je totiž $\sup_I f \leq \sup _I g$ a $f(x) \leq g(x)\leq \sup_I g(x)$.) Podle věty o limitě a uspořádání
	tedy dostáváme $\int _a^b f \leq \int_a^b g$.
\end{proof}

\begin{veta}[Aditivita Riemannova integrálu]\label{veta8.20}
Nechť $a,b,c \in \mathbb{R}$, $a<c<b$, a $f$ je funkce definovaná na $[a,b]$. Pak platí
$
f \in \mathcal R([a,b])$ právě tehdy, když $ f \in \mathcal R([a,c])$ a $f \in \mathcal R([c,b])$.
Je-li $f \in \mathcal R([a,b])$, pak
\[
\int_a^b f(x)\d x = \int_a^c f(x)\d x + \int_c^b f(x)\d x.
\]
\end{veta}

\begin{proof}
	Nalezneme posloupnost dělení $\left \{ D_n^1 \right \}_{n=1}^\infty $ intervalu $[a,c]$ splňující
	$\lim_{n \to \infty} \nu (D_n^1) = 0$. Dále nalezneme posloupnost dělení $\left \{ D_n^2 \right \}_{n=1}^\infty $
	intervalu $[c,b]$ splňující $\lim_{n \to \infty} \nu(D_n^2)=0$. Dělení $D_n$ je dělení intervalu $[a,b]$, které
	obsahuje právě dělící body $D_n^1$ a $D_n^2$. Předpokládejme, že $f\in \mathcal R([a,c])$ a
	$f \in \mathcal R([c,b])$. Chceme ukázat, že $f\in \mathcal R([a,b])$. Platí
	$$
 \lim_{n \to \infty} \overline S(f,D_n^1) = \lim_{n \to \infty} \underline S(f,D_n^1) = \int _a^c f
	$$
	podle důsledku \ref{dusledek8.13}. Obdobně
	$$
 \lim_{n \to \infty} \overline S(f,D_n^2) = \lim_{n \to \infty} \underline S(f,D_n^2) = \int _c^b f.
	$$
	Pak platí
	\begin{align*}
	\lim_{n \to \infty} \overline S(f,D_n)&=\lim_{n \to \infty} (\overline S(f,D_n^1)+ \overline S(f,D_n^2)) = \int_a^c f = \int_c^b f.\\
	\lim_{n \to \infty} \underline S(f,D_n) &= \lim_{n \to \infty} (\underline S(f,D_n^1) + \underline S(f,D_n^2)) = \int_a^b f + \int_a^b f.
\end{align*}
Proto je podle důsledku \ref{dusledek8.13} $f \in \mathcal R([a,b])$ a $\int_a^b f = \int_a^c f + \int_a^b f$.
Tím je proveden důkaz implikace zprava doleva. Zbývá dokázat, že
$$
    f \in \mathcal R([a,b]) \implies f\in \mathcal R([a,c]).
$$
Platí
\begin{align*}
0&\leq\overline S(f,D_n^1) - \underline S(f,D_n^1) \leq \overline S(f,D_n^1) - \underline S(f,D_n^1) + \underbrace{\overline S(f,D_n^2)-\underline S(f,D_n^2)}_{\leq 0}\\
&= \underbrace{\overline S(f,D_n)}_{\to \int_a^b f} - \underbrace{\underline S(f,D_n)}_{\to\int_a^b f}.
\end{align*}
Protože je
$$
\lim_{n \to \infty} \overline S (f,D_n^1) = \overline{\int_a^c} f, \qquad \lim_{n \to \infty} \underline S(f,D_n^1)=\underline{\int_a^c} f
$$
a $\lim_{n \to \infty} (\overline S(f,D_n^1) - \underline S(f,D_n^1))= 0$, máme odtud
$$
    \overline{ \int_a^c} f = \underline{\int_a^c} f ,
$$
a tedy $f \in \mathcal R([a,c])$. Analogicky je $f \in \mathcal R([c,b])$.
\end{proof}

\begin{poznamka}
	Nechť $a,b,c \in \mathbb R $. Potom platí
	$$
	    \int_a^b f + \int_b^c f + \int _c^a f=0,
	$$
	pokud alespoň dva z integrálů jsou definované.
\end{poznamka}

\begin{veta}\label{veta8.21}
Nechť $a,b \in \mathbb{R}$, $a<b$, a $f \in \mathcal R([a,b])$. Pak $|f| \in \mathcal R([a,b])$ a platí
\[
\left| \int_a^b f(x)\d x \right| \le \int_a^b |f(x)|\d x.
\]
\end{veta}

\begin{proof}
	Z $f\in \mathcal R([a,b])$ máme, že $f$ je na $[a,b]$ omezená.
	\begin{pozorovani}
		Nechť $I \subseteq [a,b]$ je neprázdný interval. Potom platí
		$$
		    \sup_I |f| - \inf_I |f| \leq \sup_I f - \inf_I f.
		$$
		Zvolme  $\varepsilon > 0$. Nalezneme $x,y\in I$  taková, že
		$$
		    |f(x)|>\sup_I |f|-\varepsilon, \qquad |f(y)| < \inf_I |f| + \varepsilon.
		$$
		Potom platí
		\begin{align*}
		\sup_I |f| - \inf_I |f| & < (|f(x)| + \varepsilon) - (|f(y)|-\varepsilon) = |f(x)| - |f(y)|+2\varepsilon \\
		&\leq |f(x)-f(y)|+ 2\varepsilon  \leq \sup_I f - \inf_I f+ 2\varepsilon,
		\end{align*}
        čímž je pozorování dokázáno.
	\end{pozorovani}
	Nechť $\mathcal D$ je dělení intervalu $[a,b]$. Potom platí
	$$
	   0\leq \overline S(|f|,\mathcal D) - \underline S(|f|,D)\leq \overline S (f,\mathcal D) - \underline S(f,\mathcal D),
	$$
	takže $f$ je riemannovsky integrovatelná díky větě \ref{veta8.14}. Zbývá dokázat nerovnost.
	\par Pro každé $x \in [a,b]$ platí
	$$
	    f(x)\leq |f(x)|.
	$$
	Odtud podle věty \ref{veta8.19} máme
	$$
	    \int_a^b f(x) \d x \leq \int _a^b |f(x)|\d x.
	$$
	Dále platí
	$$
	    -\int _a^b f \stackrel{V. \ref{veta8.18}}{=} \int_a^b (-f) \leq \int_a^b |-f| = \int_a^b |f|.
	$$
	Proto je
	$$
	    \left | \int_a^b f \right|\leq \int _a^b |f|.
	$$
\end{proof}

\begin{veta}[Derivace funkce horní meze]\label{veta8.22}
Nechť $J \subset \mathbb{R}$ je nedegenerovaný interval a $f$ je funkce definovaná na $J$ splňující
$f \in \mathcal R([a,b])$ pro každé $a,b \in J$. Nechť $c \in J$. Definujeme funkci $F$ na $J$ předpisem
\[
F(x) = \int_c^x f(t)\d t, \quad x \in J.
\]
Potom platí:
\begin{enumerate}[(a)]
	\item $F$ je spojitá na $J$,
	\item jestliže $x_0$ je vnitřním bodem intervalu $J$ a $f$ je spojitá v $x_0$, pak $F^\prime(x_0) = f(x_0)$.
\end{enumerate}
\end{veta}

\begin{proof}
	\,
	\begin{enumerate}[(a)]
		\item Nechť $x_0\in J$ a není pravým koncovým bodem $J$. Dokážeme, že
		$$
		    \lim_{x\to x_0+} F(x) = F(x_0).
		$$
		Nalezneme $\delta >0$ takové, že $x_0 + \delta \in J$. Platí, že
		$f \in \mathcal R([x_0, x_0+\delta])$. Proto lze nalézt $K>0$ takové, že
		$$
		    \forall t \in [x_0, x_0+\delta]: |f(t)|\leq K.
		$$
		Pro $x \in [x_0,x_0+\delta]$ platí
		\begin{align*}
		|F(x)-F(x_0)|  &= \left | \int_c^x f(t)\d t - \int_c^{x_0} f(t)\d t \right |\\
		& = \left |\int_{x_0}^x f(t) \d t\right| \leq \int _{x_0}^x|f(t)|\d t \leq \int_{x_0}^x K \d t = K(x-x_0)\\
		& \!\! \implies \lim_{x\to x_0+} |F(x)-F(x_0)| = 0 \implies \lim_{x\to x_0+} F(x) = F(x_0).
	    \end{align*}
		Limita zleva se dokazuje analogicky.
		\item Zvolme $\varepsilon > 0$. K němu nalezneme $\delta > 0$ takové, že
		$[x_0-\delta, x_0+\delta] \subseteq J$ a
		$$
		    \forall t \in (x_0-\delta, x_0+\delta):|f(t)-f(x_0)|<\varepsilon.
		$$
		Chceme dokázat, že
		$$
            \lim_{x\to x_0} \frac{F(x)-F(x_0)}{x-x_0}=f(x_0).
		$$
		Pro $x \in P(x_0,\delta)$ dostaneme
		\begin{align*}
		\left | \frac{F(x)-F(x_0)}{x-x_0}-f(x_0) \right | & = \left | \frac{\int_c^x f - \int_c^{x_0}f}{x-x_0}-f(x_0) \right |\\
		&= \left | \frac{\int_{x_0}^x f}{x-x_0}-f(x_0) \right | \\
		&= \frac{1}{|x-x_0|} \left | \int_{x_0}^x f(t) \d t - f(x_0)\cdot |x-x_0| \right |\\
		&= \frac{1}{|x-x_0|} \left | \int_{x_0}^x f(t) \d t - \int_{x_0}^x f(x_0)\d t \right |\\
		&= \frac{1}{|x-x_0|} \left | \int_{x_0}^x \left( f(t) - f(x_0) \right ) \d t \right |\\
	& \leq \frac{1}{|x-x_0|}\cdot |x-x_0|\cdot \varepsilon = \varepsilon.
	\end{align*}\qedhere
	\end{enumerate}
\end{proof}

\begin{dusledek}\label{dusledek8.23}\,
\begin{enumerate}[(a)]
	\item Nechť $a,b \in \mathbb{R}^\ast$, $a<b$, a $f$ je spojitá funkce na intervalu $(a,b)$. Potom $f$ má na $(a,b)$ primitivní funkci.
\item Nechť $a,b \in \mathbb{R}$, $a<b$, a $f$ je spojitá funkce na $[a,b]$ a $F$ je primitivní funkce k $f$ na $(a,b)$.
Potom existují vlastní limity $\lim_{x\to a+} F(x)$ a $\lim_{x\to b-} F(x)$ a platí
\[
\int_a^b f(t)\d t = \lim_{x\to b-} F(x) - \lim_{x\to a+} F(x).
\]
\end{enumerate}
\end{dusledek}

\begin{proof}
	\,
	\begin{enumerate}[(a)]
		\item $f$ je spojitá na intervalu $(a,b)$. Pokud $\alpha, \beta \in (a,b), \alpha < \beta$,
		potom $f \in \mathcal R([a,b])$ podle věty \ref{veta8.16}. Zvolme $c \in (a,b)$. Potom
		podle věty \ref{veta8.22}(b) platí $F^\prime(x) = f(x)$ pro každé $x \in (a,b)$, kde $F(x) = \in_c^x f$.
		$F$ je tedy primitivní funkcí k $f$ na $(a,b)$.
		\item Definujme pomocnou funkci
		$$
		    \tilde{f}(t) = \begin{cases}
							f(a), & t \in (a-1,a),\\
							f(t), & t \in [a,b], \\
							f(b), & t\in (b,b+1).
						\end{cases}
		$$
		Funkce $\tilde f$ je spojitá na intervalu $(a-1,b+1)$. Položme
		$$
		    G(x) = \int_a^x \tilde f(t) \d t, \quad x \in (a-1,b+1).
		$$
		Podle věty \ref{veta8.22}(b) dostáváme, že pro každé
		$
		    x \in (a-1,b+1)
		$
		platí $G^\prime(x) = \tilde f(x)$. Odtud plyne, že $G |_{(a,b)}$ je primitivní funkce k $f$.
		Existuje tedy $c \in \mathbb R $ takové, že $G = F+c$ na $(a,b)$. $G$ je spojitá na
		$(a-1,b+1)$. Platí tedy
		$$
  \lim_{x\to a+} G(x)=G(a), \qquad \lim_{x\to b-} G(x) = G(b).
		$$
		Potom $$\lim_{x\to a+} F(x) = \lim_{x\to a+} (G(x)-c)=G(a)-c\in \mathbb R .$$
	    Podobně
		$$
		    \lim_{x\to b-} F(x) = \lim_{x\to b-} (G(x)-c)=G(b)-c\in \mathbb R .
		$$
		Pak je
		\begin{align*}
		\int_a^b f &= G(b)=G(b)-\underbrace{G(a)}_{=\ 0}\\
	& = \left (\lim_{x\to b-} F(x)+ c\right ) - \left (\lim_{x\to a+} F(x)+c \right) \\
	&= \lim_{x\to b-} F(x) - \lim_{x\to a+} F(x).
\end{align*}\qedhere
	\end{enumerate}
\end{proof}

\begin{veta}\label{veta8.24}
Nechť $a,b \in \mathbb{R}$, $a<b$, a $f \in \mathcal R([a,b])$. Jestliže $g$ je funkce definovaná alespoň
na $[a,b]$, která se na intervalu $[a,b]$ liší od $f$ v konečném počtu bodů, pak $g \in \mathcal R([a,b])$ a
\[
\int_a^b g(x)\d x = \int_a^b f(x)\d x.
\]
\end{veta}

\begin{veta}\label{veta8.25}
Nechť $a,b \in \mathbb{R}$, $a<b$, a $f$ je funkce definovaná na intervalu $[a,b]$.
Pak jsou následující výroky ekvivalentní.
\begin{enumerate}[(i)]
	\item Funkce $f$ je Riemannovsky integrovatelná na intervalu $[a,b]$.
	\item Existuje $I \in \mathbb{R}$ takové, že pro každé $\varepsilon > 0$ existuje $\delta > 0$ splňující:
	je-li $D=\{x_i\}_{i=0}^n$ dělení intervalu $[a,b]$ takové, že $\nu(D)<\delta$ a $t_i \in [x_{i-1},x_i]$, $i=1,\dots,n$, pak
\[
\left| \sum_{i=1}^n f(t_i)(x_i - x_{i-1}) - I \right| < \varepsilon.
\]
\end{enumerate}
\end{veta}

\subsection{Newtonův integrál}
\begin{definice}
Nechť $a, b \in \mathbb{R}^\ast$, $a < b$, $f$ je funkce definovaná na intervalu $(a, b)$. Řekneme, že funkce $f$ má na intervalu $(a, b)$ \textbf{Newtonův integrál}, případně že Newtonův integrál z funkce $f$ na intervalu $(a, b)$ existuje, jestliže

\begin{itemize}
\item $f$ má na $(a, b)$ primitivní funkci $F$,
\item existují limity $\lim_{x \to a+} F(x)$ a $\lim_{x \to b-} F(x)$ (nikoli nutně vlastní),
\item rozdíl těchto dvou limit je definován jako prvek množiny $\mathbb{R}^\ast$.
\end{itemize}
\end{definice}

\begin{definice}
Hodnotou Newtonova integrálu z funkce $f$ na intervalu $(a, b)$ nazýváme prvek množiny $\mathbb{R}^\ast$ určený výrazem
\[
\lim_{x \to b-} F(x) - \lim_{x \to a+} F(x).
\]
Tuto hodnotu pak značíme symbolem
$
\int_a^b f(x)\d x.
$
Pokud $a > b$, položíme
\[
\int_a^b f(x)\d x = - \int_b^a f(x)\d x.
\]
Jestliže
$
\int_a^b f(x)\d x \in \mathbb{R},
$
pak říkáme, že Newtonův integrál z funkce $f$ na intervalu $(a, b)$ \textbf{konverguje}, v opačném případě říkáme, že \textbf{diverguje}.
\end{definice}

\begin{oznaceni}
Jestliže je potřeba rozlišit mezi Newtonovým a Riemannovým integrálem z funkce $f$ na intervalu s krajními body $a$ a $b$, kde $a, b \in \mathbb{R}$, $a < b$, budeme používat označení
\[
(N)\int_a^b f(x)\d x \qquad \text{a} \qquad (R)\int_a^b f(x)\d x.
\]
\end{oznaceni}

\begin{veta}[Linearita Newtonova integrálu]\label{veta8.26}
Nechť $a, b \in \mathbb{R}^\star$, $a < b$, a $\alpha \in \mathbb{R}$ Newtonův integrál funkcí $f, g$ existuje na $(a, b)$. Pak
\[
\int_a^b (f(x) + g(x))\d x = \int_a^b f(x)\,dx + \int_a^b g(x)\d x,
\]
pokud je výraz napravo definován. Dále platí
\[
\int_a^b \alpha f(x)\d x = \alpha \int_a^b f(x)\d x,
\]
pokud je výraz napravo definován.
\end{veta}

\begin{proof}
	Označme $F$ primitivní funkci k $f$ na $(a,b)$ a $G$
	funkci primitivní k $g$ na $(a,b)$. Pak je funkce $F+G$
	primitivní k funkci $f+g$ na $(a,b)$. Protože je
	$\left( [F]_a^b + [G]_a^b \right )\in \mathbb R ^* $,
	máme podle věty o
	aritmetice limit \ref{veta3.3}, že
$$
[F]_a^b+[G]_a^b=\left [ F+G \right ]_a^b \in \mathbb R ^* .
$$
Obdobně mějme primitivní funkci $\alpha F$ k funkci $\alpha f$
na intervalu $(a,b)$. Protože je $\alpha \cdot [F]_a^b\in R
^*$,
dostáváme, že $[\alpha F]_a^b \in R^*$ a $[\alpha F]_a^b=\alpha [F]_a^b$.
\end{proof}

\begin{veta}[Newtonův integrál a uspořádání]\label{veta8.27}
Nechť $a, b \in \mathbb{R}^\star$, $a < b$, a Newtonův integrál funkcí $f, g$ existuje na $(a, b)$. Nechť platí $f(x) \leq g(x)$ pro každé $x \in (a, b)$. Pak
\[
\int_a^b f(x)\d x \leq \int_a^b g(x)\d x.
\]
\end{veta}

\begin{proof}
	Označme funkce $F,G$ jako ve větě \ref{veta8.26}.
	Pak pro každé $x \in (a,b)$ platí
	$$
	    (G-F)^\prime(x) = g(x) - f(x) \geq 0,
	$$
	tedy $G-F$ je neklesající na $(a,b)$. Proto máme
	$$
 \lim_{x\to a+} (G(x)-G(x)) \in \mathbb R \cup \left \{ -\infty \right \}, \qquad \lim_{x\to b-} (G(x)-F(x))\in \mathbb R \cup \left \{ \infty \right \}.
	$$
	Je tedy
	$$
	    \int_a^b (g-f) = [G-F]_a^b \geq 0.
	$$
	Dále rozdělíme na dva případy.
	\begin{enumerate}[(1)]
		\item $\int_a^b f = -\infty$. Nerovnost $\int_a^b f \leq \int_a^b g$ triviálně platí.
		\item $\int_a^b f > -\infty$. Počítejme
		$$
		    \int_a^b g = \int_a^b \left( (g-f)+f \right ) = \underbrace{\int_a^b(g-f)}_{\geq \, 0}+\underbrace{\int_a^b f}_{> \, -\infty}\geq \int_a^b f,
		$$
		přičemž víme, že výraz $\int_a^b(g-f)+\int_a^b f$ je definován.\qedhere
\end{enumerate}
\end{proof}

\begin{veta}[Aditivita Newtonova integrálu]\label{veta8.28}
Nechť $a, b, c \in \mathbb{R}^\star$, $a < c < b$.

\begin{enumerate}[(a)]
\item Jestliže existuje Newtonův integrál funkce $f$ na intervalu $(a, b)$, potom existují integrály funkce $f$ na intervalech $(a, c)$ a $(c, b)$ a platí
\begin{equation}\label{integralrovnice}
	\int_a^b f(x)\d x = \int_a^c f(x)\d x + \int_c^b f(x)\d x.
\end{equation}
\item Jestliže existují Newtonovy integrály funkce $f$ na intervalech $(a, c)$ a $(c, b)$ a $f$ je spojitá v $c$, pak platí (\ref{integralrovnice}), pokud má pravá strana smysl.
\end{enumerate}
\end{veta}

\begin{proof}\,
\begin{enumerate}[(a)]
	\item Nechť $F$ je primitivní funkce k $f$ na $(a,b)$.
	Funkce $F$ je spojitá v bodě $c$, neboť $F^\prime(c)=f(c)\in \mathbb R $.
	Potom
	$$
 \lim_{x\to c+} F(x) = F(c)\in \mathbb R, \qquad \lim_{x \to c-} F(x)=F(c) \in \mathbb R.
	$$
	Dále je
	$$
	    \int_a^c f = [F]_a^c = \lim_{x\to c-} F(x)-\lim_{x\to a+} F(x)=\underbrace{F(c)}_{\in \, \mathbb R} - \underbrace{\lim_{x \to a+} F(x)}_{\in \, \mathbb R ^*},
	$$
	kde poslední člen je v $\mathbb R ^*$, neboť $[F]_a^b \in \mathbb R ^*$. Podobně i
	$\int_c^b F(x) \in R^*$. Potom
	$$
	    \int_a^c f + \int_c^b = F(c) - \lim_{x\to a+} F(x) + \lim_{x\to b-} F(x) - F(c) = \int_a^b f(x).
	$$
	\item Označme $F$ primitivní funkci k $f$ na $(a,c)$ a $G$ primitivní funkci k $f$ na $(c,b)$.
	Funkce $f$ je spojitá v $c$, takže existují $K>0$, $\delta>0$ taková, že
	$$
	    \forall x \in B(c,\delta): |f(x)|<K.
	$$
	Pro každé $x \in (c-\delta, c)$ platí
	$$
	    -K(x-c+\delta)=\int_{c-\delta}^x -K\leq\int_{c-\delta}^x f= F(x)- F(c-\delta)\leq \int_{c-\delta}^x K = K(x-c+\delta).
	$$
	Máme tedy
	$$
	    -K(x-c+\delta) + F(c-\delta) \leq F(x) \leq K(x-c+\delta) + F(c-\delta),
	$$
	a proto $\lim_{x\to c-}F(x)\in \mathbb R $ díky těmto nerovnostem a díky tomu, že víme, že limita existuje.
	Podobně  platí $\lim_{x\to c+} G(x) \in \mathbb R $. Definujme funkci $H:(a,b)\to \mathbb R $
	předpisem
	$$
	    H(x)=\begin{cases}
				F(x), & x\in (a,c),\\
				\lim_{x\to c-} F(x), & x=c,\\
				G(x)-\lim_{x\to c+} G(x) + \lim_{x\to c-} F(x), & x\in (c,b).
			\end{cases}
	$$
	Pak je $H^\prime(x) = f(x)$ pro $x \in (a,c) \cup (c,b)$. Funkce $H$ je spojitá v bodě $c$
	a $\lim_{x\to c} H^\prime(x)=\lim_{x\to c} f(x)=f(c)$. Proto platí $H^\prime(c)=f(c)$.
	Potom platí
	$$
	    [H]_a^c + [H]_c^b = [H]_a^b= \int_a^b f.
	$$\qedhere
\end{enumerate}
\end{proof}

\begin{veta}\label{veta8.29}
Nechť $a, b \in \mathbb{R}^\star$, $a < b$, a $f$ má Newtonův integrál na intervalu $(a, b)$ a $f$ je spojitá na $(a, b)$. Pak
$\int_a^b |f(x)|\d x$ existuje a
\[
 \left|\int_a^b f(x)\d x\right| \leq \int_a^b |f(x)|\d x.
\]
\end{veta}

\begin{proof}
	Označme $F$ primitivní funkci k $|f|$ na $(a,b)$ (ta existuje, neboť $|f|$ je spojitá).
	Protože je $|f| \geq 0$ a $F^\prime = |f|$, dostáváme, že $F$ je neklesající, a tedy
	$[F]_a^b$ je definována podle věty o limitě monotónní funkce \ref{veta3.9}. Navíc platí
	$[F]_a^b \geq 0$. Podle věty \ref{veta8.27} platí
	$$
	    -\int_a^b |f|\stackrel{\text{V} \ref{veta8.26}}{=}\int_a^b -|f| \leq \int_a^b f \leq \int_a^b |f|
	$$
	Dostáváme tedy $|\int_a^b f|\leq \int_a^b |f|$.
\end{proof}

\begin{veta}[Per partes pro Newtonův integrál]\label{veta8.30}
Nechť $a, b \in \mathbb{R}^\star$, $a < b$, $f$ a $g$ jsou funkce definované na $(a, b)$. Nechť $F$ je primitivní funkce k funkci $f$ na $(a, b)$ a $G$ je primitivní funkce k funkci $g$ na $(a, b)$. Potom platí
\[
\int_a^b Fg = [FG]_a^b - \int_a^b fG,
\]
jestliže má pravá strana smysl.
\end{veta}

\begin{proof}
	Označme $H$ primitivní funkci k funkci $fG$ na $(a,b)$. Počítejme
	$$
	(FG-H)^\prime = fG+Fg-fG=Fg, \qquad \text{na} (a,b).
	$$
	Pak je
	$$
	    \int_a^b Fg= [FG-H]_a^b
	$$
	a $[FG]_a^b - [H]_a^b \in \mathbb R ^*$ podle předpokladu. Podle věty o aritmetice limit
	máme
	$$
	    [FG-H]_a^b = [FG]_a^b - [H]_a^b.
	$$
\end{proof}

\begin{veta}[Substituce pro Newtonův integrál]\label{veta8.31}
Nechť $a, b, \alpha, \beta \in \mathbb{R}^\star$, $a < b$, $\alpha < \beta$, $f$ je funkce definovaná na $(a, b)$ a $\varphi$ je funkce definovaná na $(\alpha, \beta)$. Nechť $\varphi$ má vlastní nenulovou derivaci na $(\alpha, \beta)$ a platí $\varphi((\alpha, \beta)) = (a, b)$. Potom
\[
\int_a^b f(x)\d x = \int_\alpha^\beta f(\varphi(t))\cdot |\varphi^\prime(t)|\d t,
\]
jestliže má alespoň jedna strana smysl.
\end{veta}

\begin{proof}
	Funkce $\varphi^\prime$ má Darbouxovu vlastnost, a proto nemění znaménko na $(\alpha,\beta)$.
	Odtud plyne, že $\varphi$ je ryze monotónní. Nejprve předpokládejme, že $\varphi$ je klesající.
	Předpokládejme dále, že $\int_a^b f$ existuje. Označme $F$ primitivní funkci k funkci $f$ na intervalu
	$(a,b)$. Potom $-F\circ \varphi$ je primitivní k funkci $-f\circ \varphi \cdot \varphi^\prime=f\circ \varphi |\varphi^\prime|$.
	Platí $\lim_{t\to \alpha +} \varphi(t) = b$ a $\lim_{t\to \beta -}\varphi(t) = a$.
	Dále je
	$$
	\lim_{t\to \alpha+} (-F\circ\varphi(t))=-\lim_{x\to b-} F(x)
	$$
	podle věty o limitě složené funkce s podmínkou (P). Obdobně
	$$
 \lim_{t\to \beta-} (-F\circ\varphi(t))=-\lim_{x\to a+} F(x).
	$$
	Nakonec dostáváme
	\begin{align*}
	\int_\alpha^\beta f(\varphi(t))|\varphi^\prime (t)|\d t & = \int _\alpha ^\beta f(\varphi(t))(-\varphi^\prime(t))\d t = [-F(\varphi(t))]_\alpha ^\beta \\
	&= -\lim_{x\to a+} F(x) + \lim_{x\to b-} F(x) = [F]_a^b = \int_a^b f.
	\end{align*}
	\par Nyní předpokládejme, že $\int_\alpha^\beta f(\varphi(t))|\varphi^\prime(t)|\d t$ existuje. Nechť
	$G$ je primitivní funkce k funkci $f\circ \varphi \cdot |\varphi^\prime| = -f\circ \varphi \cdot\varphi^\prime$.
	Potom $-G\circ \varphi^{-1}$ je primitivní k funkci $f$ na $(a,b)$ podle věty \ref{veta8.7}.
	Potom platí $\lim_{x\to a+}\varphi^{-1}(x)=\beta$ a $\lim_{x\to b-}\varphi^{-1}(x)=\alpha$. Dále je
	$\lim_{x\to a+} -G(\varphi^{-1}(x))=-\lim_{t\to \beta-}G(t)$ a $\lim_{x\to b-}-G(\varphi^{-1}(x)) = -\lim_{t\to \alpha+} G(t)$.
	Konečně máme
	$$
	    \int_a^b f(x) \d x = [-G\circ \varphi^{-1}]_a^b = -\lim_{t\to \alpha+} G(t) + \lim_{t\to \beta -} G(t) = \int_a^b f(\varphi(t)) |\varphi^\prime(t)|\d t.
	$$
	Případ $\varphi$ rostoucí dokážeme analogicky.
\end{proof}

\begin{veta}[Bolzanova--Cauchyova podmínka pro funkce]\label{veta8.32}
Nechť $a \in \mathbb{R}^\star$, $\delta_0 > 0$, a funkce $F$ je definována alespoň na $P(a, \delta_0)$.
Potom existuje vlastní $\lim_{x \to a} F(x)$ právě tehdy, když je splněna následující (tzv. Bolzanova--Cauchyova) podmínka:
\[
\forall \varepsilon > 0 \, \exists \delta > 0 \, \forall x, y \in P(a, \delta): |F(x) - F(y)| < \varepsilon.
\]
\end{veta}
